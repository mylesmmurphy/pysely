from typing import Generic, Literal, Never, Protocol, Self, TypeVar, overload, reveal_type

from test.fixtures.portable_schema import Database

Table = Literal["person", "pet"]
T = TypeVar("T", bound=Table)
Q = TypeVar("Q", bound=Table, covariant=True)

class Query(Generic[Q]):
    def where(self, column: Literal["first_name"], value: object) -> Self:
        return self

class Baseline(Protocol):
    @overload
    def join(self, table: Literal["person"]) -> Query[Literal["person"]]: ...
    @overload
    def join(self, table: Literal["pet"]) -> Query[Literal["pet"]]: ...

class Reordered(Protocol):
    @overload
    def join(self, table: Literal["pet"]) -> Query[Literal["pet"]]: ...
    @overload
    def join(self, table: Literal["person"]) -> Query[Literal["person"]]: ...

class LiteralFallback(Protocol):
    @overload
    def join(self, table: Literal["person"]) -> Query[Literal["person"]]: ...
    @overload
    def join(self, table: Literal["pet"]) -> Query[Literal["pet"]]: ...
    @overload
    def join(self, table: Table) -> Query[Table]: ...

class BoundFallback(Protocol):
    @overload
    def join(self, table: Literal["person"]) -> Query[Literal["person"]]: ...
    @overload
    def join(self, table: Literal["pet"]) -> Query[Literal["pet"]]: ...
    @overload
    def join(self, table: T) -> Query[T]: ...

class SameReturn(Protocol):
    @overload
    def join(self, table: Literal["person"]) -> Query[Table]: ...
    @overload
    def join(self, table: Literal["pet"]) -> Query[Table]: ...

class BroadFallback(Protocol):
    @overload
    def join(self, table: Literal["person"]) -> Query[Literal["person"]]: ...
    @overload
    def join(self, table: Literal["pet"]) -> Query[Literal["pet"]]: ...
    @overload
    def join(self, table: str) -> Query[Table]: ...

class NeverFallback(Protocol):
    @overload
    def join(self, table: Literal["person"]) -> Query[Literal["person"]]: ...
    @overload
    def join(self, table: Literal["pet"]) -> Query[Literal["pet"]]: ...
    @overload
    def join(self, table: str) -> Never: ...

class Single(Protocol):
    def join(self, table: Table) -> Query[Table]: ...

class SingleGeneric(Protocol):
    def join(self, table: T) -> Query[T]: ...

class ConcreteImplementation:
    @overload
    def join(self, table: Literal["person"]) -> Query[Literal["person"]]: ...
    @overload
    def join(self, table: Literal["pet"]) -> Query[Literal["pet"]]: ...
    def join(self, table: str) -> Query[Table]:
        return Query()

def baseline(db: Baseline) -> None:
    reveal_type(db.join("pett").where("first_name", "Jennifer"))

def reordered(db: Reordered) -> None:
    reveal_type(db.join("pett").where("first_name", "Jennifer"))

def literal_fallback(db: LiteralFallback) -> None:
    reveal_type(db.join("pett").where("first_name", "Jennifer"))

def bound_fallback(db: BoundFallback) -> None:
    reveal_type(db.join("pett").where("first_name", "Jennifer"))

def same_return(db: SameReturn) -> None:
    reveal_type(db.join("pett").where("first_name", "Jennifer"))

def broad_fallback(db: BroadFallback) -> None:
    reveal_type(db.join("pett").where("first_name", "Jennifer"))

def never_fallback(db: NeverFallback) -> None:
    reveal_type(db.join("pett").where("first_name", "Jennifer"))

def single(db: Single) -> None:
    reveal_type(db.join("pett").where("first_name", "Jennifer"))

def single_generic(db: SingleGeneric) -> None:
    reveal_type(db.join("pett").where("first_name", "Jennifer"))

def bad_column(query: Query[Table]) -> None:
    reveal_type(query.where("first_namee", "Jennifer"))

def concrete_implementation(db: ConcreteImplementation) -> None:
    reveal_type(db.join("pett").where("first_name", "Jennifer"))

def actual_where(db: Database) -> None:
    reveal_type(db.select_from("person").where("first_namee", "=", "Jennifer").select("first_name"))

def actual_join(db: Database) -> None:
    reveal_type(db.select_from("person").inner_join("pett", "owner_id", "person.id").where("first_name", "=", "Jennifer"))
